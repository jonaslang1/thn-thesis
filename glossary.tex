\newglossaryentry{Build}{
    name=Build,
    description={
        Der Prozess, bei dem der Quellcode in ein ausführbares Format umgewandelt wird.
        Dies kann das Kompilieren von Code, das Zusammenstellen von Ressourcen und das Erstellen von Installationspaketen umfassen
    },
}

\newglossaryentry{Deployment}{
    name=Deployment,
    description={
        Der Prozess, bei dem eine Anwendung oder ein System in einer Produktionsumgebung bereitgestellt wird.
        Dies umfasst die Installation, Konfiguration und Aktivierung der Software auf den Zielsystemen
    },
}

\newglossaryentry{Mock}{
    name=Mock,
    description={
        Ein Testdoppel, das das Verhalten eines echten Objekts simuliert.
    },
}

\newglossaryentry{Stakeholder}{
    name=Stakeholder,
    description={
        Eine Person oder Gruppe, die ein Interesse an einem Projekt hat und von dessen Ergebnissen betroffen ist.
        Stakeholder können Kunden, Benutzer, Entwickler, Manager und andere Beteiligte sein
    },
}

\newglossaryentry{DevOps}{
    name=DevOps,
    description={
        Eine Kombination aus Entwicklung (Dev) und Betrieb (Ops), die darauf abzielt, die Zusammenarbeit zwischen Softwareentwicklern und IT-Betriebsteams zu verbessern.
        DevOps fördert Automatisierung, kontinuierliche Integration und kontinuierliche Bereitstellung
    },
}

\newglossaryentry{Pipeline}{
    name=Pipeline,
    description={
        Eine Abfolge von automatisierten Schritten, die zur Erstellung, Prüfung und Bereitstellung von Software verwendet werden.
        Pipelines sind ein zentraler Bestandteil von \gls{CI}/\gls{CD}-Workflows
    },
}

\newglossaryentry{OpenAPI}{
    name=OpenAPI,
    description={
        Eine Spezifikation für \gls{REST}ful \gls{API}s, die es Entwicklern ermöglicht, \gls{API}s zu beschreiben, zu erstellen und zu dokumentieren.
        \gls{OpenAPI} verwendet eine standardisierte Syntax, um \gls{API}-Endpunkte, Parameter und Datenformate zu definieren
    },
}

\newglossaryentry{AsyncAPI}{
    name=AsyncAPI,
    description={
        Eine Spezifikation für asynchrone \gls{API}s, die es Entwicklern ermöglicht, asynchrone Nachrichtenprotokolle zu beschreiben und zu dokumentieren.
        \gls{AsyncAPI} ist besonders nützlich für Systeme, die auf Ereignissen basieren
    },
}

\newglossaryentry{Spring}{
    name=Spring,
    description={
        Ein weit verbreitetes Framework für die Entwicklung von Java-Anwendungen, das eine Vielzahl von Modulen und Funktionen bietet, um die Entwicklung zu erleichtern.
        \gls{Spring} ist besonders bekannt für seine Unterstützung von \gls{API}-Entwicklung und Microservices
    },
}

\newglossaryentry{Maven}{
    name=Maven,
    description={
        Ein Build-Management-Tool für Java-Projekte, das die Verwaltung von Abhängigkeiten, das Erstellen von Artefakten und die Ausführung von Tests automatisiert.
        \gls{Maven} ist ein wichtiger Bestandteil des Java-Ökosystems
    },
}

\newglossaryentry{Angular}{
    name=Angular,
    description={
        Ein weit verbreitetes Framework für die Entwicklung von Webanwendungen, das auf TypeScript basiert und eine komponentenbasierte Architektur bietet.
        \gls{Angular} wird häufig für die Entwicklung von Single-Page-Anwendungen verwendet
    },
}

\newglossaryentry{Docker}{
    name=Docker,
    description={
        Eine Plattform zur Automatisierung der Bereitstellung von Anwendungen in Containern, die eine konsistente und isolierte Ausführungsumgebung bieten.
        \gls{Docker} ermöglicht es Entwicklern, Anwendungen und ihre Abhängigkeiten in Containern zu verpacken und zu verteilen
    },
}

\newglossaryentry{NoSQL}{
    name=NoSQL,
    description={
        Eine Klasse von Datenbanken, die nicht auf dem relationalen Modell basieren und für unstrukturierte oder semi-strukturierte Daten geeignet sind.
        \gls{NoSQL}-Datenbanken bieten Flexibilität und Skalierbarkeit für moderne Anwendungen
    },
}
